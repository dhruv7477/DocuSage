#   docker run --rm -v "$PWD":/repo:ro -w /repo python:3.12-slim sh -c '
#     pip install -q --upgrade pip &&
#     pip install -q "$(grep "^torch==" requirements.txt)" --index-url https://download.pytorch.org/whl/cpu &&
